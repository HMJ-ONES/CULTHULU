cmake_minimum_required(VERSION 3.16)
project(CultCanonicalCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "${REPO}/src/*.cpp")
# Standalone drivers/fuzz harnesses have their own main and POSIX-only helpers.
list(FILTER CORE_SOURCES EXCLUDE REGEX "/(driver|fuzz)/")
list(FILTER CORE_SOURCES EXCLUDE REGEX "/sim/main.cpp$")
add_library(cultulhu STATIC ${CORE_SOURCES})
target_include_directories(cultulhu PUBLIC "${REPO}/src")
target_compile_definitions(cultulhu PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES)
if(WIN32)
 target_link_libraries(cultulhu PUBLIC ws2_32 iphlpapi)
endif()
enable_testing()
foreach(SUITE wave7a wave7b wave7c wave7d wave7e wave25 wave26 wave27 beta wave2 wave3 wave4 content_wave9 directives_wave9 wave15 wave28 wave29 belief_binding)
 add_executable(test_${SUITE} "${REPO}/tests/tests_${SUITE}.cpp")
 target_link_libraries(test_${SUITE} PRIVATE cultulhu)
 add_test(NAME ${SUITE} COMMAND test_${SUITE})
 set_tests_properties(${SUITE} PROPERTIES WORKING_DIRECTORY "${REPO}")
endforeach()
